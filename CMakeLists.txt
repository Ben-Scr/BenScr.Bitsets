cmake_minimum_required(VERSION 3.28)
project(BenScrBitsets VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
endif()

option(BENSCR_BITSETS_BUILD_TESTS "Build contract tests" ${PROJECT_IS_TOP_LEVEL})
option(BENSCR_BITSETS_BUILD_EXAMPLES "Build example" ${PROJECT_IS_TOP_LEVEL})
option(BENSCR_BITSETS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(benscr_bitsets INTERFACE)
add_library(BenScr::Bitsets ALIAS benscr_bitsets)
set_target_properties(benscr_bitsets PROPERTIES EXPORT_NAME Bitsets)
target_compile_features(benscr_bitsets INTERFACE cxx_std_20)
target_include_directories(benscr_bitsets INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

function(benscr_bitsets_checks target)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(BENSCR_BITSETS_SANITIZERS)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        else()
            message(FATAL_ERROR "Sanitizers require GCC or Clang on a supported POSIX host.")
        endif()
    endif()
endfunction()

if(BUILD_TESTING AND BENSCR_BITSETS_BUILD_TESTS)
    enable_testing()
    add_executable(bitsets_contract_tests tests/bitsets_tests.cpp tests/allocation_failures.cpp)
    target_link_libraries(bitsets_contract_tests PRIVATE BenScr::Bitsets)
    benscr_bitsets_checks(bitsets_contract_tests)
    add_test(NAME bitsets_contract COMMAND bitsets_contract_tests)
    add_executable(bitsets_header_test tests/header.cpp tests/header_peer.cpp)
    target_link_libraries(bitsets_header_test PRIVATE BenScr::Bitsets)
    benscr_bitsets_checks(bitsets_header_test)
    add_test(NAME standalone_header COMMAND bitsets_header_test)
endif()
if(BENSCR_BITSETS_BUILD_EXAMPLES)
    add_executable(bitsets_example examples/bitsets.cpp)
    target_link_libraries(bitsets_example PRIVATE BenScr::Bitsets)
    benscr_bitsets_checks(bitsets_example)
endif()

install(TARGETS benscr_bitsets EXPORT BenScrBitsetsTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE.txt README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/BenScrBitsets)
configure_package_config_file(cmake/BenScrBitsetsConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/BenScrBitsetsConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrBitsets)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/BenScrBitsetsConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion ARCH_INDEPENDENT)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BenScrBitsetsConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/BenScrBitsetsConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrBitsets)
install(EXPORT BenScrBitsetsTargets NAMESPACE BenScr:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrBitsets)
